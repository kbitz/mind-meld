## API surface (final)

```python
# src/mind_meld/events.py
from typing import TypedDict
from concurrent.futures import as_completed, ThreadPoolExecutor

INITIAL_CURSOR_LOOKBACK_DAYS    = 30
WALK_TIME_BUDGET_INTERACTIVE_MS = 500
WALK_TIME_BUDGET_AUTOPUSH_MS    = 250
MAX_GIT_WORKERS                 = 8
PER_REPO_TIMEOUT_FLOOR_MS       = 200
PER_REPO_TIMEOUT_CAP_MS         = 2000

# CT-8: 'del' is a Python reserved word. Use functional TypedDict so the JSON
# field is "del", not "del_".
GitCommit = TypedDict("GitCommit", {
    "sha": str,
    "date": str,           # %cI commit date (CT-13: matches `git log --since` filter)
    "author_email": str,
    "subject": str,
    "files": int,
    "add": int,
    "del": int,
}, total=False)

class GitSnapshotProject(TypedDict, total=False):
    remote: str            # canonicalize_remote_url() output (no credentials)
    local_path: str
    commits: list[GitCommit]

class GitSnapshotSkip(TypedDict, total=False):
    path: str
    reason: str

class GitSnapshot(TypedDict, total=False):
    v: int                 # 1
    type: str              # "git-snapshot"
    ts: str
    device: str
    projects: list[GitSnapshotProject]
    skipped: list[GitSnapshotSkip]   # A5: per-repo failures fold here

class SessionMetadata(TypedDict, total=False):
    claude_dir: str
    sessions: int
    total_kb: int
    last_session_at: str
    ephemeral: bool

class SessionsSnapshot(TypedDict, total=False):
    v: int
    type: str              # "sessions-snapshot"
    ts: str
    device: str
    projects: list[SessionMetadata]

class MmPushEvent(TypedDict, total=False):
    v: int
    type: str              # "mm-push"
    ts: str
    device: str
    mm_version: str
    sources: list[dict]                 # CT-9 ack: untyped for v=1
    discovery_errors: list[str]         # CT-11: forensic trail for failed probers


def canonicalize_remote_url(url: str) -> str:
    """Normalize git remote URL to <host>/<org>/<repo> form.

    SECURITY (CT-10): MUST strip ALL of the following before producing canonical
    form:
      - userinfo segment (`user:password@host` → strip `user:password@`)
      - x-access-token / OAuth tokens
      - query-string credentials (`?token=...`, `?access_token=...`)
      - URL fragments
    Lowercase host; preserve path case; strip trailing `.git`.
    Test must include adversarial credential-laden inputs."""

def discover_git_roots(config: dict) -> tuple[list[Path], list[str]]:
    """Multi-prober registry. Returns (roots, errors).

    Probers (deduped at end via Path.resolve()):
      1. gstack:  ~/.gstack/projects/*/repo-mode.json (when gstack source enabled)
      2. claude:  decode ~/.claude/projects/<encoded>/ → fs paths (when claude source enabled)
      3. manual:  config["retro"]["repo_roots"]: list[str]
      4. (future) codex: dispatch by source name when codex source lands

    Filter (CT-1): keep paths where `git rev-parse --show-toplevel` succeeds.
    Worktrees have `.git` as a FILE, not a directory — Conductor workspaces are
    worktrees. The earlier .git-subdir check would have silently excluded them.

    Per-prober failures append to `errors: list[str]` and emit `mm: notice:`.
    Empty registry returns ([], []) — distinguishable from broken-discovery
    via the `errors` list (CT-11)."""

def walk_git_projects(roots: list[Path], since: datetime, total_budget_ms: int) -> list[GitSnapshot]:
    """Capture commits since `since` from each root. Bounded by total_budget_ms.

    Wall-time enforcement (CT-6): submit all repos to ThreadPoolExecutor, then
    pump `as_completed(futures, timeout=remaining_budget_s)`. On TimeoutError,
    cancel pending futures (running ones can't be cancelled but are orphaned;
    process exit cleans them up). Collect completed results.

    Per-repo timeout (A2):
        max(PER_REPO_TIMEOUT_FLOOR_MS,
            (total_budget_ms * MAX_GIT_WORKERS) // repos_remaining)
        capped at PER_REPO_TIMEOUT_CAP_MS

    Subprocess (CT-13): record `%cI` (commit date) so `git log --since` filter
    matches the retro render window:
        git -C <root> log --since=<iso> --numstat -M -C \\
            --format='%x1e%H%x09%cI%x09%ae%x09%s'

    Per-repo failures fold into row's `skipped: [{path, reason}]` (A5).
    Whole-walk failure: `mm: notice:` forensic, return what completed."""

def walk_session_metadata(claude_dir: Path, since: datetime) -> list[SessionsSnapshot]:
    """os.scandir 2-level walk of ~/.claude/projects/<encoded>/. Sets
    `ephemeral: True` when decoded path matches */conductor/workspaces/* (path-
    string match, NOT existence check). Perf <500ms for 10k files."""

def last_push_ts(events_dir: Path, device_id: str) -> datetime:
    """Cursor derivation (A3 + A4 elimination of cursor file).

    Reverse-scan up to INITIAL_CURSOR_LOOKBACK_DAYS daily files for this device,
    return ts of latest `{"type":"mm-push", ...}` line. Default to
    `now - INITIAL_CURSOR_LOOKBACK_DAYS` if not found.

    Pattern matches pullhistory.py: log file IS the state of truth.
    No separate cursor file. No separate flock domain. No lock-order rule."""

def write_push_event(events_dir: Path, device_id: str, events: list[dict]) -> None:
    """Append N events to today's per-device JSONL via fsutil.flock_append_jsonl.

    Order invariant (CT-4): caller MUST construct events with mm-push event
    LAST. Partial write before mm-push appends → cursor doesn't advance →
    next push re-walks the range (deduped at retro render via (remote, sha)).
    Single flock window is best-effort batching, NOT transactionality.

    Pre-write: mkdir(events_dir, parents=True, exist_ok=True) — config bootstrap
    creates the BASE source path; the events/ subdir is bootstrapped only when
    get_sources fires. Track 7A doesn't rely on that ordering (CT-12).

    File mode 0o600. Per-day naming: events/<device>-<YYYY-MM-DD>.jsonl."""
```

## fsutil.py — extended

```python
def flock_append_jsonl(
    path: Path,
    lines: list[bytes],   # one JSON-encoded row per element, NO trailing newline
    *,
    mode: int = 0o600,
    on_locked: Callable[[int], None] | None = None,  # called under flock with the fd
) -> None:
    """Append N JSONL rows to `path` atomically under fcntl.flock(LOCK_EX).

    Contract (CT-5): blocking LOCK_EX, no retry/timeout. Matches pullhistory.py
    semantics (mm.lockfile already serializes push-vs-push, so contention is
    rare). Callers who need retry budget compose around this helper; the helper
    itself stays simple.

    Best-effort: OSError swallowed (callers are forensic logs, not data
    integrity).

    `on_locked(fd)` runs under flock AFTER the writes; pullhistory uses this
    closure for line-boundary rotation."""
```

`pullhistory._append_payload` retrofits to call `flock_append_jsonl(path, [line], on_locked=rotate_under_lock_if_over_cap)`. Existing pullhistory tests must still pass (regression check).

## Section 1 architecture decisions (Reviewer A1–A5)

| Decision | Result |
|---|---|
| **A1** discover_git_roots roots source | Multi-prober registry: gstack + claude + manual config; codex prober deferred until source lands. |
| **A2** Per-repo timeout vs total budget | `max(200, (total_budget_ms * 8) // repos_remaining)`, cap 2000ms. |
| **A3** Cursor flock semantics | **ELIMINATED.** Cursor derived from events.jsonl tail via `last_push_ts`. No cursor file, no flock domain, no lock-order rule. |
| **A4** Cursor/events write order | **ELIMINATED.** No separate cursor write. mm-push event in events.jsonl IS the cursor. |
| **A5** Per-repo skip event type | Skips fold into git-snapshot row's `skipped` field. Schema stays at 3 event types. |

### Bucket T0 — design's original 6 cases
- URL canonicalization round-trip table (5 forms × edge cases)

### Bucket T1 — new surface from Section 1+2 decisions
- **discover_git_roots**:
  - gstack prober reads `~/.gstack/projects/*/repo-mode.json`
  - claude prober decodes encoded dir → filesystem path
  - manual `[retro].repo_roots` additive
  - dedup via `Path.resolve()` (APFS case-mismatch)
  - filter via `git rev-parse --show-toplevel` (CT-1: worktrees pass, plain dirs fail, .git-as-file passes)
  - gstack source disabled → no gstack probe
  - claude source disabled → no claude probe
  - empty registry → `([], [])`
- **last_push_ts** (replaces cursor tests):
  - first run (no events files) → `now - 30d`
  - finds latest mm-push line in today's file
  - walks backward through prior days when today empty
  - 30-day lookback boundary
- **fsutil.flock_append_jsonl**:
  - multi-line atomic append under flock
  - `mode` parameter respected
  - `on_locked` callback fires under flock with fd
- **pullhistory retrofit**:
  - all existing pullhistory tests still pass (regression)
  - rotation behavior preserved (closure over `on_locked`)

### Bucket T2 — correctness pins for review decisions
- walk_git_projects timeout formula (30 repos × 250ms × 8 workers; verify wall time bounded by `as_completed` at the pump)
- Budget abort mid-walk: mm-push event written LAST, partial git events kept, next push's `last_push_ts` returns prior-push timestamp (cursor doesn't advance on partial)
- as_completed pump cancels pending futures on TimeoutError (CT-6)

### Bucket T3 — codex CT-corrections + hardening
- canonicalize_remote_url adversarial inputs:
  - `https://x-access-token:TOKEN@host/foo.git` → `host/foo` (token stripped)
  - `https://user:pass@host/foo.git` → `host/foo` (userinfo stripped)
  - `https://host/foo.git?token=abc` → `host/foo` (query auth stripped)
  - `https://host/foo.git#fragment` → `host/foo` (fragment stripped)
  - empty `""` and malformed `"foo"` inputs (defined behavior)
  - GitLab subgroup paths preserved (`host/group/subgroup/repo`)
- write_push_event:
  - file mode 0o600
  - per-day naming `events/<device>-<YYYY-MM-DD>.jsonl`
  - mkdir `events_dir` before first append (CT-12)
  - all N events in single flock window (atomicity batching)
  - mm-push event MUST be last in events list (CT-4)
- walk_git_projects:
  - records `%cI` (commit date) in `date` field, not `%aI` (CT-13)
  - subprocess TimeoutExpired per-repo → fold into `skipped`
  - empty roots `[]` → `[]`
  - repo where `git rev-parse --show-toplevel` fails → silently skipped
  - repo with no commits since `since` → empty `commits` list
- walk_session_metadata:
  - empty claude_dir → `[]`
  - non-jsonl files ignored
  - `total_kb` summing
  - `last_session_at = max(mtime)` of all jsonl files

## Failure modes

| Failure | Behavior | User-facing |
|---|---|---|
| gstack probe fails | Skip prober, append to `errors`; emit `mm: notice:` | Forensic trail in mm-push event's `discovery_errors` |
| claude probe fails | Same as above | Same |
| All probers fail + manual empty | Return `([], errors)`; walk_git_projects([]) → []; no git events | retro shows "0 commits"; `discovery_errors` distinguishes "broken" from "clean" |
| Subprocess TimeoutExpired per-repo | Fold into row's `skipped` field | Forensic, retro continues |
| Whole-walk failure (executor crash, etc.) | `mm: notice:`; return events accumulated so far | Push continues |
| events.jsonl flock contention (rare) | Blocking flock waits; mm.lockfile serializes higher up | Negligible latency |
| events.jsonl write failure mid-batch | mm-push event was last → cursor doesn't advance → next push re-walks | Deduped at retro render via `(remote, sha)` |
| events_dir missing | mkdir on first write (CT-12) | Silent success |

## Track 7B prerequisite (CT-3 ack)

Track 7B's `_push_core` wiring MUST run the events tail on every push attempt,
even when `_push_core` early-returns on no-content-diff. Without this, machines
that push regularly but rarely change content silently never advance their
cursor and never appear in fleet retros.

Implementation hint: events tail belongs INSIDE the locked region of
`_push_core`, AFTER `_ensure_device_registered` and BEFORE the no-content-diff
early return.
